视觉表现：

卡片动作：卡片可能会 360度翻转（Flip）或者剧烈闪烁彩虹光泽。

目标图片：兴奋地跳跃并旋转，带有残影效果。

粒子特效：全屏礼花爆炸（Confetti Explosion）。屏幕两侧喷射出彩带，粒子数量巨大（100+），且下落速度变慢，营造漫天飞舞的感觉。

背景：背景色瞬间闪过一道暖光（如淡金色或彩虹渐变），营造高光时刻。

2. 粒子系统 (Particle System)

这部分代码基于 HTML5 Canvas，实现了一个支持不同“等级”的粒子爆炸效果。

/* ==========================================
   PARTICLE SYSTEM (粒子系统)
   功能：根据传入的 level 参数，生成不同数量、形状和速度的粒子
   ========================================== */

const canvas = document.getElementById('confetti-canvas');
const ctx = canvas.getContext('2d');
let particles = [];

// 1. 绘制五角星的辅助函数
function drawStar(cx, cy, spikes, outerRadius, innerRadius, color) {
    let rot = Math.PI / 2 * 3;
    let x = cx;
    let y = cy;
    let step = Math.PI / spikes;

    ctx.beginPath();
    ctx.moveTo(cx, cy - outerRadius);
    for (let i = 0; i < spikes; i++) {
        x = cx + Math.cos(rot) * outerRadius;
        y = cy + Math.sin(rot) * outerRadius;
        ctx.lineTo(x, y);
        rot += step;

        x = cx + Math.cos(rot) * innerRadius;
        y = cy + Math.sin(rot) * innerRadius;
        ctx.lineTo(x, y);
        rot += step;
    }
    ctx.lineTo(cx, cy - outerRadius);
    ctx.closePath();
    ctx.fillStyle = color;
    ctx.fill();
}

// 2. 创建爆炸效果 (核心逻辑)
// x, y: 爆炸中心坐标
// level: 反馈等级 (1=弱, 2=中, 3=强)
function createExplosion(x, y, level = 1) {
    const colors = ['#ff595e', '#ffca3a', '#8ac926', '#1982c4', '#6a4c93', '#ff924c'];
    
    // 默认参数 (Level 1)
    let count = 15; // 粒子数量
    let speed = 8;  // 扩散速度
    
    // 根据等级调整参数
    if (level === 2) { 
        count = 40; 
        speed = 12; 
    }
    if (level === 3) { 
        count = 100; 
        speed = 15; 
        // Level 3 强制改为屏幕中心爆炸 (全屏庆祝)
        x = window.innerWidth / 2; 
        y = window.innerHeight / 2; 
    } 

    // 生成粒子
    for (let i = 0; i < count; i++) {
        // Level 2 以上有 30% 概率生成星星形状，否则为圆形
        const isStar = level >= 2 && Math.random() > 0.7; 
        
        particles.push({
            x: x,
            y: y,
            vx: (Math.random() - 0.5) * speed, // X轴速度
            vy: (Math.random() - 0.5) * speed, // Y轴速度
            radius: Math.random() * (level * 3) + 2, // 等级越高粒子越大
            color: colors[Math.floor(Math.random() * colors.length)],
            life: 1, // 生命值 (透明度)
            decay: level === 3 ? 0.01 : 0.03, // Level 3 消失得更慢 (留存时间长)
            type: isStar ? 'star' : 'circle',
            gravity: level === 3 ? 0.2 : 0 // Level 3 增加重力感，模拟礼花下落
        });
    }
}

// 3. 动画循环
function animateParticles() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += p.gravity || 0; // 应用重力
        p.life -= p.decay;      // 减少生命值
        
        if (p.type !== 'star') p.radius *= 0.96; // 圆形粒子逐渐变小

        ctx.globalAlpha = p.life;
        
        if (p.type === 'star') {
            drawStar(p.x, p.y, 5, p.radius, p.radius/2, p.color);
        } else {
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
            ctx.fillStyle = p.color;
            ctx.fill();
        }
        
        ctx.globalAlpha = 1;

        // 移除死掉的粒子
        if (p.life <= 0) {
            particles.splice(i, 1);
        }
    }
    requestAnimationFrame(animateParticles);
}
// 启动循环
animateParticles();


3. JS 交互控制逻辑 (Interaction Logic)

这是在点击选项卡片时触发的代码，它将上述的 CSS 动画和 Canvas 粒子结合在一起。

注意：在当前版本中，为了最大化反馈感，我们在视觉上强制使用了 Level 3 的效果。

/* 函数：selectOption
   参数：
     - selectedOption: 数据对象
     - cardElement: 被点击的 DOM 元素
     - event: 点击事件
*/
selectOption(selectedOption, cardElement, event) {
  // 1. 基础校验 (游戏是否激活、是否正在播放动画、是否点在错误卡片上)
  if (!this.isGameActive || this.isProcessing) return;
  if (cardElement.classList.contains('wrong')) return;

  const isCorrect = selectedOption.id === this.currentTarget.id;
  
  if (isCorrect) {
      // === 答对逻辑 ===
      
      this.isProcessing = true; // 锁定交互，防止连点，预留动画时间
      this.correctAnswers++;
      this.consecutiveCorrect++; // 连击数 +1

      /* --- 策略层：计算理论反馈等级 ---
         虽然当前视觉强制 Level 3，但保留此逻辑方便后续恢复多级反馈
      */
      let feedbackLevel = 1;
      if (this.consecutiveCorrect >= 2 && this.consecutiveCorrect < 5) feedbackLevel = 2;
      if (this.consecutiveCorrect >= 5) feedbackLevel = 3;

      /* --- 表现层：执行视觉效果 ---
         当前设置：强制 visualLevel = 3 (终极反馈)
      */
      const visualLevel = 3; 

      // 1. 触发 CSS 卡片动画
      cardElement.classList.remove('hint-highlight'); // 移除提示光圈
      cardElement.classList.add('correct-lvl3');    // 添加终极动画类 (3D翻转)

      // 2. 触发 CSS 目标图片动画
      const targetFrame = document.getElementById('targetFrame');
      targetFrame.classList.add('bounce-big');      // 添加大幅跳动

      // 3. 触发 Canvas 粒子特效
      const rect = cardElement.getBoundingClientRect();
      const centerX = rect.left + rect.width / 2;
      const centerY = rect.top + rect.height / 2;
      // 传入 visualLevel 控制粒子数量和行为
      createExplosion(centerX, centerY, visualLevel); 

      // 4. 触发背景闪光特效 (Level 3 特有)
      document.body.classList.add('flash-bg');

      // 5. 时间轴控制 (Timing)
      // 延迟 2000ms 后进入下一题，给动画和(未来的)语音留出展示时间
      setTimeout(() => {
          this.nextQuestion();
      }, 2000); 

  } else {
      // === 答错逻辑 (保持简约) ===
      this.consecutiveCorrect = 0; // 连击清零
      cardElement.classList.add('wrong'); // 简单的抖动变灰
  }
}
